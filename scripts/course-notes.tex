% call with Libertinus Mono instead of editing the vendored class.

% A week opens with "Week" over the big numeral in the margin and, under the
% heading, that week's own contents. kaobook resets the chapter style at
% \mainmatter, so the numeral format is applied per chapter here.
\renewcommand*{\chaptermarkformat}{Week~\thechapter\enskip}
\newcommand*{\notesweek}[2]{%
  \setcounter{chapter}{\numexpr#1-1\relax}%
  \renewcommand*{\chapterformat}{\mbox{\parbox[b]{\marginparwidth}{\raggedright
    {\large\itshape Week}\\[-1pt]\scalebox{2.85}{\thechapter}}}}%
  \setchapterpreamble[u]{\margintoc}%
  \chapter{#2}}
\unsettoc{toc}{totoc}

% Definitions and examples count within the week: Definition 1.1, 1.2, ...
\counterwithout{definition}{section}\counterwithin{definition}{chapter}
\counterwithout{example}{section}\counterwithin{example}{chapter}

% Blocks the theorem package does not have, in the same framed style.
\newcommand*{\notesname}[2]{\if\relax\detokenize{#1}\relax#2\else#1\fi}
\newtcolorbox{keyidea}[1][]{tcbkao, colback=keybg,
  before upper={\textbf{\notesname{#1}{Key idea.}}\enspace}}
\newtcolorbox{caution}[1][]{tcbkao, colback=cautionbg,
  before upper={\textbf{\textcolor{caution}{\notesname{#1}{Caution.}}}\enspace}}
% Side notes, like the sticky notes on the handwritten page, go in the margin.
\NewDocumentEnvironment{aside}{o +b}
  {\marginnote{\vspace{.5\baselineskip}\raggedright\hyphenpenalty=10000\IfValueT{#1}{\textbf{#1}\enspace}#2}}{}
\newcommand*{\notesbreak}{\par\medskip\noindent\hfil\rule{.25\linewidth}{.4pt}\hfil\par\medskip}

% The week's key ideas again, gathered by the converter before its exercises.
\newtcolorbox{notesrecap}{tcbkao, colback=keybg, borderline west={2pt}{0pt}{keyrule},
  before upper={\textbf{Key points of the week}\par\smallskip}}

% Exercises are numbered within the week and answered at the back of the book;
% each links to its answer and each answer links back.
\newcounter{notesexercise}[chapter]
\renewcommand*{\thenotesexercise}{\thechapter.\arabic{notesexercise}}
\newtcolorbox{notesexercise}[1]{tcbkao, colback=examplebg,
  before upper={\refstepcounter{notesexercise}\label{#1}\textbf{Exercise~\thenotesexercise.}\enspace},
  after upper={\par\smallskip\hfill{\small\hyperref[#1-answer]{\color{quiet}Answer on page~\pageref*{#1-answer}}}}}
\newcommand{\notesanswers}[1]{%
  \chapter*{Answers to the exercises}\addcontentsline{toc}{chapter}{Answers to the exercises}#1}
\newcommand{\notesanswer}[2]{%
  \par\noindent\textbf{\hyperref[#1]{Exercise~\ref*{#1}}}\label{#1-answer}\enspace#2\par\medskip}

% A small chip in a block's own colour, for the reading guide.
\newcommand*{\notesswatch}[2]{\tcbox[on line, colback=#1, colframe=#1, boxrule=0pt, sharp corners,
  boxsep=0pt, left=4pt, right=4pt, top=2pt, bottom=2pt]{\small #2}}

% Cover. Ink-blue, with the dot grid of the notebook the notes are written in.
% Faint sample paths of a random walk and one realised path in gold: week 1 in
% a picture. The walks are seeded from the course, so a course keeps its cover.
\begin{titlepage}
  \begin{tikzpicture}[remember picture, overlay]
    \fill[coverink] (current page.south west) rectangle (current page.north east);
    \begin{scope}[shift={(current page.south west)}]
      \foreach \x in {1,...,20} \foreach \y in {1,...,29} \fill[white, opacity=.07] (\x,\y) circle (.6pt);
      \pgfmathsetseed{<<seed>>}
      \foreach \i in {1,...,9}{
        \draw[white, opacity=.13, line width=.5pt, line join=round]
          (1.5,8.5) \foreach \s in {1,...,300}{ -- ++(.06,{.16*rand}) };}
      \draw[covergold, line width=1.3pt, line join=round]
        (1.5,8.5) \foreach \s in {1,...,300}{ -- ++(.06,{.16*rand}) };
      \fill[covergold] (1.5,8.5) circle (2pt);
    \end{scope}
    % A narrow measure keeps a long title to balanced lines.
    \node[anchor=north west, align=flush left, text width=14cm, text=white]
        at ([xshift=2cm, yshift=-2.6cm]current page.north west) {%
      {\large\color{covergold}<<code>>}\\[1.2cm]
      {\fontsize{54}{58}\selectfont\hyphenpenalty=10000 <<name>>\par}\vspace{.7cm}
      {\Large\color{white!70!coverink}Lecture notes, <<term>>}};
    \node[anchor=south west, align=flush left, text width=12cm, text=white!75!coverink]
        at ([xshift=2cm, yshift=2.2cm]current page.south west) {%
      {\large\color{white}<<author>>}\\[1mm]Middle East Technical University, Ankara};
    \node[anchor=south east, text=white!55!coverink]
        at ([xshift=-2cm, yshift=2.2cm]current page.south east) {%
      \small\hypersetup{urlcolor=white!55!coverink}\href{<<url>>}{<<site>>}};
  \end{tikzpicture}
\end{titlepage}

\chapter*{How to read these notes}
These are my notes from <<code>>, typeset from the pages I write in class and
updated as the term goes on. Each week is a chapter, and the margin at the
start of a week lists what it covers.

\medskip
{\raggedright % chips cannot stretch, so let the line end where it ends
\notesswatch{defbg}{Definitions} are numbered within the week.
\notesswatch{examplebg}{Examples and exercises} are grey.
\notesswatch{keybg}{Key ideas} are amber, and
\notesswatch{cautionbg}{\color{caution}points to be careful about} are red.
Asides from the margin of my notebook stay in the margin.\par}

\medskip
Every week closes with its key points and a few exercises. The answers are at
the back, followed by an index of the terms in bold.
