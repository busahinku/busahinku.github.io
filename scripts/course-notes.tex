% 46 faint sample paths and one realised path (#1) across the cover. Called
% twice with the same seed, so the paths line up across the band's edge.
\newcommand*{\notespaths}[1]{%
  \foreach \i in {1,...,46}{%
    \pgfmathsetmacro{\shade}{25+50*rnd}%
    \draw[accent!\shade!white, opacity=.4, line width=.45pt, line join=round]
      (-.2,{10.2+1.2*rand}) \foreach \x in {1,...,215}{ -- ++(.1,{.22*rand}) };}%
  \draw[#1, line width=1.5pt, line join=round]
    (-.2,10.2) \foreach \x in {1,...,215}{ -- ++(.1,{.22*rand}) };}

% Weeks are chapters. The margin carries "Week" over the big numeral and,
% below the heading, that week's own contents.
% A week opens with "Week" over the big numeral in the margin and, under the
% heading, that week's own contents. kaobook resets the chapter style at
% \mainmatter, so the numeral format is applied per chapter here.
\renewcommand*{\chaptermarkformat}{Week~\thechapter\enskip}
\newcommand*{\notesweek}[2]{%
  \setcounter{chapter}{\numexpr#1-1\relax}%
  \renewcommand*{\chapterformat}{\mbox{\parbox[b]{\marginparwidth}{\raggedright
    {\large\itshape Week}\\[-1pt]\scalebox{2.85}{\thechapter}}}}%
  \setchapterpreamble[u]{\margintoc}%
  \chapter{#2}}
\unsettoc{toc}{totoc}

% Definitions and examples count within the week: Definition 1.1, 1.2, ...
\counterwithout{definition}{section}\counterwithin{definition}{chapter}
\counterwithout{example}{section}\counterwithin{example}{chapter}

% Blocks the theorem package does not have, in the same framed style.
\newcommand*{\notesname}[2]{\if\relax\detokenize{#1}\relax#2\else#1\fi}
\newtcolorbox{keyidea}[1][]{tcbkao, colback=keybg,
  before upper={\textbf{\notesname{#1}{Key idea.}}\enspace}}
\newtcolorbox{caution}[1][]{tcbkao, colback=cautionbg,
  before upper={\textbf{\textcolor{caution}{\notesname{#1}{Caution.}}}\enspace}}
\newtcolorbox{selfcheck}[1]{tcbkao, colback=examplebg,
  before upper={\textbf{Check.}\enspace #1\par\smallskip\textit{Answer.}\enspace}}

% Cover. The bottom of the page is a tilted band of random-walk sample paths,
% one realised path picked out; above the band the same paths run on and fade
% into the paper. The walks are seeded from the course, so a course keeps its
% cover. Drop a logo at scripts/cover-logo.pdf and it sits above the university.
\begin{titlepage}
  \begin{tikzpicture}[remember picture, overlay]
    \begin{scope}[shift={(current page.south west)}]
      \def\band{(0,0) -- (0,8.4) -- (21,10.2) -- (21,0) -- cycle}
      \def\above{(0,8.4) -- (21,10.2) -- (21,16) -- (0,16) -- cycle}
      \fill[coverink] \band;
      \begin{scope}
        \clip \above;
        \pgfmathsetseed{<<seed>>}\notespaths{accent}
        \fill[white, path fading=south] (0,8.4) rectangle (21,16.1);
      \end{scope}
      \begin{scope}
        \clip \band;
        \pgfmathsetseed{<<seed>>}\notespaths{white}
      \end{scope}
    \end{scope}

    \node[anchor=north, align=center] at ([yshift=-3cm]current page.north) {%
      \IfFileExists{cover-logo.pdf}{\includegraphics[height=2.2cm]{cover-logo.pdf}\\[5mm]}{}%
      {\small\bfseries Middle East Technical University}};
    \node[align=flush center, text width=16cm] at ([yshift=-10.6cm]current page.north) {%
      {\small\bfseries <<code>>, lecture notes}\\[4mm]
      {\fontsize{34}{41}\selectfont\itshape <<name>>\par}\vspace{4mm}
      {\large <<term>>}};
    \node[align=center] at ([yshift=-14.6cm]current page.north) {%
      {\small\bfseries Notes by}\\[1.5mm]{\large <<author>>}};
    \node[align=center, text=white] at ([yshift=3cm]current page.south) {%
      Ankara \textbar{} Last updated <<updated>>\\[2mm]
      {\footnotesize\hypersetup{urlcolor=white!70!coverink}\url{<<url>>}}};
  \end{tikzpicture}
\end{titlepage}
