% Notes blocks: a hairline on the left and a run-in label, the way a printed
% textbook marks them. Definitions and examples are numbered within the week.
% A title replaces the label; the converter gives it its full stop.
\newcommand*{\notestitle}[1]{\if\relax\detokenize{#1}\relax\else\ (#1)\fi}
\newcommand*{\notesname}[2]{\if\relax\detokenize{#1}\relax#2\else#1\fi}
\tcbset{notes/.style={blanker, breakable, parbox=false, left=11pt, right=0pt,
  top=1pt, bottom=1pt, borderline west={.9pt}{0pt}{#1},
  before skip=.9\baselineskip, after skip=.9\baselineskip}}
\newtcolorbox[auto counter, number within=chapter]{definition}[1][]{notes=accent,
  before upper={\textbf{Definition~\thetcbcounter\notestitle{#1}.}\enspace}}
\newtcolorbox[auto counter, number within=chapter]{example}[1][]{notes=hair,
  before upper={\textbf{Example~\thetcbcounter\notestitle{#1}.}\enspace}}
\newtcolorbox{keyidea}[1][]{notes=accent,
  before upper={\textbf{\notesname{#1}{Key idea.}}\enspace}}
\newtcolorbox{caution}[1][]{notes=caution,
  before upper={\textbf{\notesname{#1}{Caution.}}\enspace}}
\newtcolorbox{aside}[1][]{notes=hair, fontupper=\small,
  before upper={\textbf{\notesname{#1}{Note.}}\enspace}}
\newtcolorbox{selfcheck}[1]{notes=hair,
  before upper={\textbf{Check.}\enspace #1\par\smallskip\textit{Answer.}\enspace}}
